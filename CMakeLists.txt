cmake_minimum_required(VERSION 3.16)
project(MiniLangCompiler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(minilang src/main.cpp)
if(MSVC)
  target_compile_options(minilang PRIVATE /W4)
else()
  target_compile_options(minilang PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME arithmetic COMMAND minilang ${CMAKE_CURRENT_SOURCE_DIR}/examples/arithmetic.mini)
